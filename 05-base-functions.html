10 Apply Family

m <- matrix(1:9, nrow=3)
apply(m, 1, mean)  # row means
[1] 4 5 6
apply(m, 2, mean)  # col means
[1] 2 5 8
lst <- list(a=1:3, b=10:12)
sapply(lst, mean)   # simplifies result
 a  b 
 2 11 
mapply(sum, 1:3, 10:12)
[1] 11 13 15
